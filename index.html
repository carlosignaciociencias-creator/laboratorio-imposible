import cv2 import mediapipe as mp import math import random import time # ============================================================ # LABORATORIO IMPOSIBLE # Tres experimentos historicos sobre radiacion # Control mediante movimientos de las manos # ============================================================ # ----------------------------- # MEDIA PIPE # ----------------------------- mp_hands = mp.solutions.hands mp_draw = mp.solutions.drawing_utils hands = mp_hands.Hands( static_image_mode=False, max_num_hands=2, min_detection_confidence=0.55, min_tracking_confidence=0.55 ) # ----------------------------- # CAMARA # ----------------------------- cap = cv2.VideoCapture(0) if not cap.isOpened(): print("No se pudo abrir la camara.") print("Verifica que otra aplicacion no este utilizando la camara.") exit() # ----------------------------- # VARIABLES GENERALES # ----------------------------- experimento = 0 ultimo_click = 0 # Variables del experimento 1 tiempo_exposicion = 5 radiacion_1 = 50 # Variables del experimento 2 angulo = 0 energia = 50 desviadas = 0 # Variables del experimento 3 distancia = 50 intensidad = 50 detecciones = 0 # ============================================================ # FUNCIONES # ============================================================ def distancia_puntos(p1, p2): return math.sqrt( (p1[0] - p2[0]) ** 2 + (p1[1] - p2[1]) ** 2 ) def dedo_indice_arriba(landmarks): """ Comprueba si el dedo indice esta extendido. """ return landmarks[8][1] < landmarks[6][1] def pinza(landmarks): """ Detecta gesto de pinza: punta del pulgar cerca de punta del indice. """ d = distancia_puntos( landmarks[4], landmarks[8] ) return d < 45 def boton(x, y, w, h, texto, frame, activo=False): """ Dibuja un boton. """ color = (70, 180, 255) if activo else (60, 60, 60) cv2.rectangle( frame, (x, y), (x + w, y + h), color, -1 ) cv2.rectangle( frame, (x, y), (x + w, y + h), (255, 255, 255), 2 ) cv2.putText( frame, texto, (x + 15, y + 32), cv2.FONT_HERSHEY_SIMPLEX, 0.65, (255, 255, 255), 2 ) def barra(frame, x, y, w, valor, minimo, maximo, etiqueta): """ Dibuja una barra para visualizar una variable. """ cv2.putText( frame, etiqueta, (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) cv2.rectangle( frame, (x, y), (x + w, y + 25), (60, 60, 60), -1 ) proporcion = (valor - minimo) / (maximo - minimo) ancho = int(w * proporcion) cv2.rectangle( frame, (x, y), (x + ancho, y + 25), (70, 180, 255), -1 ) cv2.putText( frame, str(int(valor)), (x + w + 10, y + 20), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) def escribir_centro(frame, texto, y, escala=0.8): tamaño = cv2.getTextSize( texto, cv2.FONT_HERSHEY_SIMPLEX, escala, 2 )[0] x = (frame.shape[1] - tamaño[0]) // 2 cv2.putText( frame, texto, (x, y), cv2.FONT_HERSHEY_SIMPLEX, escala, (255, 255, 255), 2 ) # ============================================================ # EXPERIMENTO 1 # BECQUEREL - PLACA FOTOGRAFICA # ============================================================ def experimento_1(frame): global tiempo_exposicion global radiacion_1 escribir_centro( frame, "EXPERIMENTO 1 - PLACA FOTOGRAFICA", 55, 0.75 ) cv2.putText( frame, "Explora que ocurre al cambiar el tiempo de exposicion.", (35, 95), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (220, 220, 220), 1 ) # Placa placa_x = 90 placa_y = 180 placa_w = 350 placa_h = 230 cv2.rectangle( frame, (placa_x, placa_y), (placa_x + placa_w, placa_y + placa_h), (180, 180, 180), -1 ) # Oscurecimiento segun exposicion osc = int( min( 180, tiempo_exposicion * radiacion_1 / 2 ) ) cv2.rectangle( frame, (placa_x + 10, placa_y + 10), (placa_x + placa_w - 10, placa_y + placa_h - 10), (max(30, 220 - osc), max(30, 220 - osc), max(30, 220 - osc)), -1 ) # Fuente radiactiva virtual cv2.circle( frame, (600, 290), 55, (70, 100, 255), -1 ) cv2.putText( frame, "FUENTE", (555, 300), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2 ) # Rayos for i in range(5): cv2.line( frame, (555, 290 + (i - 2) * 20), (440, 290 + (i - 2) * 20), (70, 220, 255), 2 ) barra( frame, 520, 420, 250, tiempo_exposicion, 1, 10, "TIEMPO" ) barra( frame, 520, 480, 250, radiacion_1, 10, 100, "INTENSIDAD" ) cv2.putText( frame, "PINZA + MOVIMIENTO HORIZONTAL = modificar variables", (40, 520), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) # Resultado if tiempo_exposicion * radiacion_1 > 300: resultado = "La placa presenta una exposicion intensa." elif tiempo_exposicion * radiacion_1 > 150: resultado = "La placa presenta una exposicion intermedia." else: resultado = "La placa presenta una exposicion baja." cv2.putText( frame, resultado, (40, 570), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (100, 255, 180), 2 ) # ============================================================ # EXPERIMENTO 2 # RUTHERFORD - DISPERSION # ============================================================ def experimento_2(frame): global angulo global energia global desviadas escribir_centro( frame, "EXPERIMENTO 2 - DISPERSION DE PARTICULAS", 55, 0.72 ) cv2.putText( frame, "Modifica energia y observa como cambia la trayectoria.", (35, 95), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (220, 220, 220), 1 ) # Pantalla centro_x = 610 centro_y = 300 cv2.circle( frame, (centro_x, centro_y), 55, (80, 180, 100), -1 ) cv2.putText( frame, "NUCLEO", (570, 305), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 2 ) # Particulas random.seed(10) desviadas = 0 for i in range(18): y_inicio = 150 + i * 18 x_inicio = 100 dx = centro_x - x_inicio dy = centro_y - y_inicio factor = 0.45 x_final = centro_x + int( dx * factor ) y_final = centro_y + int( dy * factor ) # Probabilidad de desviacion if abs(y_inicio - centro_y) < 70: desviacion = (50 - energia) * 1.2 y_final += int( desviacion * ((i % 3) - 1) ) desviadas += 1 cv2.line( frame, (x_inicio, y_inicio), (x_final, y_final), (80, 220, 255), 2 ) barra( frame, 80, 500, 250, energia, 10, 100, "ENERGIA" ) barra( frame, 450, 500, 250, abs(angulo) + 1, 1, 90, "ANGULO" ) cv2.putText( frame, "Particulas desviadas: " + str(desviadas), (80, 565), cv2.FONT_HERSHEY_SIMPLEX, 0.65, (100, 255, 180), 2 ) cv2.putText( frame, "Observa la relacion entre energia, trayectoria e interaccion.", (80, 610), cv2.FONT_HERSHEY_SIMPLEX, 0.52, (255, 255, 255), 1 ) # ============================================================ # EXPERIMENTO 3 # MEDICION DE RADIACION # ============================================================ def experimento_3(frame): global distancia global intensidad global detecciones escribir_centro( frame, "EXPERIMENTO 3 - DETECCION Y DISTANCIA", 55, 0.75 ) cv2.putText( frame, "Explora como cambia la medicion al alejar el detector.", (35, 95), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (220, 220, 220), 1 ) # Fuente cv2.circle( frame, (350, 300), 60, (70, 100, 255), -1 ) cv2.putText( frame, "FUENTE", (305, 305), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 2 ) # Detector detector_x = 350 + int(distancia * 4) cv2.rectangle( frame, (detector_x - 35, 250), (detector_x + 35, 350), (80, 180, 220), -1 ) cv2.putText( frame, "DETECTOR", (detector_x - 45, 380), cv2.FONT_HERSHEY_SIMPLEX, 0.45, (255, 255, 255), 1 ) # Intensidad inversamente relacionada con distancia intensidad_calculada = max( 5, min( 100, 10000 / (distancia * distancia + 100) ) ) barra( frame, 80, 470, 300, distancia, 5, 100, "DISTANCIA" ) barra( frame, 500, 470, 250, intensidad_calculada, 5, 100, "MEDICION" ) cv2.putText( frame, "Pregunta: que relacion observas entre distancia y medicion?", (70, 550), cv2.FONT_HERSHEY_SIMPLEX, 0.58, (100, 255, 180), 2 ) cv2.putText( frame, "No se utiliza una fuente real: es una simulacion segura.", (70, 595), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (220, 220, 220), 1 ) # ============================================================ # MENU # ============================================================ def menu(frame): escribir_centro( frame, "LABORATORIO IMPOSIBLE", 80, 1.0 ) escribir_centro( frame, "RADIACION A TRAVES DE LA HISTORIA", 125, 0.7 ) boton( 180, 200, 600, 65, "1 - PLACA FOTOGRAFICA", frame ) boton( 180, 290, 600, 65, "2 - DISPERSION DE PARTICULAS", frame ) boton( 180, 380, 600, 65, "3 - DETECCION Y DISTANCIA", frame ) cv2.putText( frame, "Usa el dedo indice para seleccionar.", (270, 490), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 255, 255), 2 ) cv2.putText( frame, "Pulsa Q para salir.", (350, 540), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (200, 200, 200), 1 ) # ============================================================ # BUCLE PRINCIPAL # ============================================================ while True: ret, frame = cap.read() if not ret: break frame = cv2.flip(frame, 1) rgb = cv2.cvtColor( frame, cv2.COLOR_BGR2RGB ) resultado = hands.process(rgb) # -------------------------------------------------------- # DETECCION DE MANOS # -------------------------------------------------------- dedo_x = None dedo_y = None haciendo_pinza = False if resultado.multi_hand_landmarks: for mano in resultado.multi_hand_landmarks: mp_draw.draw_landmarks( frame, mano, mp_hands.HAND_CONNECTIONS ) puntos = [] for punto in mano.landmark: puntos.append( ( int(punto.x * frame.shape[1]), int(punto.y * frame.shape[0]) ) ) if dedo_indice_arriba(puntos): dedo_x = puntos[8][0] dedo_y = puntos[8][1] if pinza(puntos): haciendo_pinza = True # -------------------------------------------------------- # MENU # -------------------------------------------------------- if experimento == 0: menu(frame) if ( dedo_x is not None and dedo_y is not None and haciendo_pinza ): ahora = time.time() if ahora - ultimo_click > 1: if 200 < dedo_y < 265: experimento = 1 elif 290 < dedo_y < 355: experimento = 2 elif 380 < dedo_y < 445: experimento = 3 ultimo_click = ahora # -------------------------------------------------------- # EXPERIMENTO 1 # -------------------------------------------------------- elif experimento == 1: experimento_1(frame) cv2.putText( frame, "M - volver al menu", (30, 650), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) # Movimiento de la mano modifica el tiempo if dedo_x is not None: tiempo_exposicion = int( max( 1, min( 10, dedo_x / frame.shape[1] * 10 ) ) ) # -------------------------------------------------------- # EXPERIMENTO 2 # -------------------------------------------------------- elif experimento == 2: experimento_2(frame) cv2.putText( frame, "M - volver al menu", (30, 650), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) if dedo_x is not None: energia = int( max( 10, min( 100, dedo_x / frame.shape[1] * 100 ) ) ) angulo = int( (dedo_y if dedo_y else 300) / frame.shape[0] * 90 ) # -------------------------------------------------------- # EXPERIMENTO 3 # -------------------------------------------------------- elif experimento == 3: experimento_3(frame) cv2.putText( frame, "M - volver al menu", (30, 650), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (255, 255, 255), 1 ) if dedo_x is not None: distancia = int( max( 5, min( 100, dedo_x / frame.shape[1] * 100 ) ) ) # -------------------------------------------------------- # VOLVER AL MENU # -------------------------------------------------------- tecla = cv2.waitKey(1) & 0xFF if tecla == ord("q"): break if tecla == ord("m"): experimento = 0 # ============================================================ # FINALIZAR # ============================================================ cap.release() hands.close() cv2.destroyAllWindows()
